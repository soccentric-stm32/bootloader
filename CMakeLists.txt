# Target:  STM32F767ZIT6
# SRS:     SRS-BOOT-F767 (BCON-001..003, BCON-005, BGEN-001, LAY-001, BNFR-001,
#          BSAFE-006)
#
# Build (inside the stm32-dev container, BCON-001):
#   cmake -S . -B build -G Ninja
#   cmake --build build
# or simply: scripts/build.sh [clean]
#
# Outputs in build/: boot.elf, boot.bin, boot.hex, boot.map (BGEN-001).
#

cmake_minimum_required(VERSION 3.22)

set(BOOT_VERSION "0.1.0")

# Cross toolchain: the build runs in the container where arm-none-eabi-gcc
# is the only compiler. Set explicitly so a stray host compiler is never used.
set(CMAKE_SYSTEM_NAME     Generic)
set(CMAKE_SYSTEM_PROCESSOR arm)
set(CMAKE_TRY_COMPILE_TARGET_TYPE STATIC_LIBRARY) # no OS to run a test exe in
set(CMAKE_EXPORT_COMPILE_COMMANDS ON) # compile_commands.json for clang-tidy (make lint)
find_program(ARMCC arm-none-eabi-gcc REQUIRED)
find_program(ARMOBJCOPY arm-none-eabi-objcopy REQUIRED)
find_program(SIZE_TOOL arm-none-eabi-size REQUIRED)
set(CMAKE_C_COMPILER   ${ARMCC})
set(CMAKE_ASM_COMPILER ${ARMCC})
set(CMAKE_OBJCOPY      ${ARMOBJCOPY})

project(boot LANGUAGES C ASM)

# --- SDK location ----------------------------------------------------------
# STM32CubeF7 v1.17.4 is provided by the docker image (BCON-002).
set(CUBE_ROOT "$ENV{STM32CUBE_F7}")
if(NOT CUBE_ROOT)
    set(CUBE_ROOT "/opt/st/STM32CubeF7")
endif()
if(NOT EXISTS "${CUBE_ROOT}/Drivers/STM32F7xx_HAL_Driver/Inc/stm32f7xx_hal.h")
    message(FATAL_ERROR
        "STM32CubeF7 not found at ${CUBE_ROOT}.\n"
        "Build through scripts/docker-dev.sh (see docker/README.md).")
endif()

# --- Build metadata (BGEN-002) ---------------------------------------------
execute_process(COMMAND git -C ${CMAKE_CURRENT_SOURCE_DIR} rev-parse --short=12 HEAD
    OUTPUT_VARIABLE BOOT_GIT_COMMIT OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
if(NOT BOOT_GIT_COMMIT)
    set(BOOT_GIT_COMMIT "unknown")
endif()
string(TIMESTAMP BOOT_BUILD_DATE "%Y-%m-%d %H:%M:%S" UTC)

# --- Toolchain flags --------------------------------------------------------
set(CPU_FLAGS -mcpu=cortex-m7 -mthumb -mfpu=fpv5-d16 -mfloat-abi=hard)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)

# Size-optimised in every build type: BOOT must fit in 32 KB (BCON-005).
# Link-time optimisation saves about 4 KB, which pays for the signature
# check (BSAFE-006).
add_compile_options(${CPU_FLAGS} -Wall -Wextra -Werror -ffunction-sections -fdata-sections -flto)
# flatcc runtime allocation hooks (BCON-003: no malloc) are visible wherever
# the runtime headers expand FLATCC_ALLOC.  Applied to C sources only.
set(CMAKE_C_FLAGS "${CMAKE_C_FLAGS} -include ${CMAKE_SOURCE_DIR}/src/protocol/flatcc_port.h")
# BCON-003: no dynamic memory and no stdio, so no syscall layer is linked.
add_link_options(${CPU_FLAGS} -flto -Os -Wl,--gc-sections -Wl,--cref
    -Wl,-Map=${CMAKE_BINARY_DIR}/boot.map
    --specs=nano.specs)

if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release)
endif()
set(CMAKE_C_FLAGS_RELEASE "-Os -g3")
set(CMAKE_C_FLAGS_DEBUG   "-Og -g3")

# --- Image authentication (BSAFE-006) ----------------------------------------
# BOOT_SIGNED_IMAGES=ON accepts only version 2 images signed with the key in
# BOOT_SIGNING_KEY_SOURCE (generated by tools/mkimage.py --export-pubkey).
# The default is the published development key; a product build must pass
# its own, e.g. -DBOOT_SIGNING_KEY_SOURCE=/secure/product_pubkey.c.
option(BOOT_SIGNED_IMAGES "Accept only signed images (BSAFE-006)" ON)
set(BOOT_DEV_KEY_SOURCE "${CMAKE_SOURCE_DIR}/keys/dev_pubkey.c")
set(BOOT_SIGNING_KEY_SOURCE "${BOOT_DEV_KEY_SOURCE}" CACHE FILEPATH
    "C definition of the image signing public key")
if(NOT EXISTS "${BOOT_SIGNING_KEY_SOURCE}")
    message(FATAL_ERROR "BOOT_SIGNING_KEY_SOURCE not found: ${BOOT_SIGNING_KEY_SOURCE}")
endif()
if(BOOT_SIGNED_IMAGES)
    get_filename_component(_key_real "${BOOT_SIGNING_KEY_SOURCE}" REALPATH)
    get_filename_component(_dev_real "${BOOT_DEV_KEY_SOURCE}" REALPATH)
    if(_key_real STREQUAL _dev_real)
        message(WARNING
            "BOOT accepts images signed with the published development key "
            "(keys/dev-signing-key.pem).  Do not ship this build; set "
            "BOOT_SIGNING_KEY_SOURCE to the product key.")
    endif()
    set(BOOT_SIGNED_IMAGES_VALUE 1)
else()
    message(WARNING "BOOT_SIGNED_IMAGES=OFF: unsigned images are accepted.")
    set(BOOT_SIGNED_IMAGES_VALUE 0)
endif()

add_compile_definitions(
    BOOT_SIGNED_IMAGES=${BOOT_SIGNED_IMAGES_VALUE}
    # micro-ecc: verify on secp256r1 only, Cortex-M7 UMAAL assembly.
    uECC_SUPPORTS_secp160r1=0
    uECC_SUPPORTS_secp192r1=0
    uECC_SUPPORTS_secp224r1=0
    uECC_SUPPORTS_secp256k1=0
    uECC_SUPPORT_COMPRESSED_POINT=0
    uECC_OPTIMIZATION_LEVEL=1
    uECC_PLATFORM=uECC_arm_thumb2
    uECC_ARM_USE_UMAAL=1
    STM32F767xx
    USE_HAL_DRIVER
    HSE_VALUE=8000000U          # ST-LINK MCO feeds PH0 in bypass mode
    BOOT_VERSION_STR="${BOOT_VERSION}"
    BOOT_SRS_ID="SRS-BOOT-F767"
    BOOT_GIT_COMMIT="${BOOT_GIT_COMMIT}"
    BOOT_BUILD_DATE="${BOOT_BUILD_DATE}")

# Source tree layout (see src/README.md):
#   app/      boot flow: main, slot selection, update state machine, hand-over
#   bsp/      STM32F7 board support (HAL-facing drivers)
#   common/   hardware-independent helpers: layout, CRC32, SHA-256, image, metadata
#   protocol/ HWLINK wire protocol, flatcc port and generated FlatBuffers headers
include_directories(
    src/app
    src/bsp
    src/common
    src/protocol
    board
    "${CMAKE_SOURCE_DIR}/src/protocol/generated"
    "${CMAKE_SOURCE_DIR}/third_party/flatcc/include"
    "${CMAKE_SOURCE_DIR}/third_party/micro-ecc"
    "${CUBE_ROOT}/Drivers/CMSIS/Include"
    "${CUBE_ROOT}/Drivers/CMSIS/Device/ST/STM32F7xx/Include"
    "${CUBE_ROOT}/Drivers/STM32F7xx_HAL_Driver/Inc")

# --- Sources ----------------------------------------------------------------
# HAL modules used so far. The list grows with each SRS section.
set(HAL_SOURCES
    "${CUBE_ROOT}/Drivers/STM32F7xx_HAL_Driver/Src/stm32f7xx_hal.c"
    "${CUBE_ROOT}/Drivers/STM32F7xx_HAL_Driver/Src/stm32f7xx_hal_cortex.c"
    "${CUBE_ROOT}/Drivers/STM32F7xx_HAL_Driver/Src/stm32f7xx_hal_rcc.c"
    "${CUBE_ROOT}/Drivers/STM32F7xx_HAL_Driver/Src/stm32f7xx_hal_rcc_ex.c"
    "${CUBE_ROOT}/Drivers/STM32F7xx_HAL_Driver/Src/stm32f7xx_hal_gpio.c"
    "${CUBE_ROOT}/Drivers/STM32F7xx_HAL_Driver/Src/stm32f7xx_hal_pwr.c"
    "${CUBE_ROOT}/Drivers/STM32F7xx_HAL_Driver/Src/stm32f7xx_hal_pwr_ex.c"
    "${CUBE_ROOT}/Drivers/STM32F7xx_HAL_Driver/Src/stm32f7xx_hal_flash.c"
    "${CUBE_ROOT}/Drivers/STM32F7xx_HAL_Driver/Src/stm32f7xx_hal_flash_ex.c"
    "${CUBE_ROOT}/Drivers/STM32F7xx_HAL_Driver/Src/stm32f7xx_hal_dma.c"
    "${CUBE_ROOT}/Drivers/STM32F7xx_HAL_Driver/Src/stm32f7xx_hal_dma_ex.c"
    "${CUBE_ROOT}/Drivers/STM32F7xx_HAL_Driver/Src/stm32f7xx_hal_uart.c"
    "${CUBE_ROOT}/Drivers/STM32F7xx_HAL_Driver/Src/stm32f7xx_hal_uart_ex.c")

# The generated flatcc headers use compact macro formatting that trips
# -Wmisleading-indentation in the one translation unit that includes them.
set_source_files_properties(src/protocol/proto.c PROPERTIES
    COMPILE_OPTIONS "-Wno-misleading-indentation")

# flatcc 0.6.1 runtime as released (BCON-002).  Only builder, verifier,
# emitter and refmap are linked; JSON support is not used.
set(FLATCC_ROOT "${CMAKE_SOURCE_DIR}/third_party/flatcc")
set(FLATCC_SOURCES
    "${FLATCC_ROOT}/src/runtime/builder.c"
    "${FLATCC_ROOT}/src/runtime/verifier.c"
    "${FLATCC_ROOT}/src/runtime/emitter.c"
    "${FLATCC_ROOT}/src/runtime/refmap.c")

# app: boot flow
set(APP_SOURCES
    src/app/main.c
    src/app/handover.c
    src/app/selection.c
    src/app/update.c
    src/app/swap_jump.S)

# bsp: STM32F7 board support
set(BSP_SOURCES
    src/bsp/clock.c
    src/bsp/fault.c
    src/bsp/flash.c
    src/bsp/iwdg.c
    src/bsp/uart3.c)

# common: hardware-independent helpers
set(COMMON_SOURCES
    src/common/crc32.c
    src/common/image.c
    src/common/metadata.c
    src/common/sha256.c)

# protocol: HWLINK server and flatcc integration
set(PROTOCOL_SOURCES
    src/protocol/hwlink.c
    src/protocol/proto.c
    src/protocol/flatcc_port.c)

set(PROJECT_SOURCES
    ${APP_SOURCES}
    ${BSP_SOURCES}
    ${COMMON_SOURCES}
    ${PROTOCOL_SOURCES})

set(DEVICE_SOURCES
    "${CUBE_ROOT}/Drivers/CMSIS/Device/ST/STM32F7xx/Source/Templates/system_stm32f7xx.c"
    "${CUBE_ROOT}/Drivers/CMSIS/Device/ST/STM32F7xx/Source/Templates/gcc/startup_stm32f767xx.s")

# micro-ecc (BSD-2-Clause) as released; only uECC_verify() is linked.
set(UECC_SOURCES)
if(BOOT_SIGNED_IMAGES)
    set(UECC_SOURCES "${CMAKE_SOURCE_DIR}/third_party/micro-ecc/uECC.c")
endif()

add_executable(boot.elf ${PROJECT_SOURCES} ${HAL_SOURCES} ${DEVICE_SOURCES}
    ${FLATCC_SOURCES} ${UECC_SOURCES} "${BOOT_SIGNING_KEY_SOURCE}")
target_compile_definitions(boot.elf PRIVATE
    NDEBUG=1
    BOOT_FLATCC_HEAP_SIZE=8192
    FLATCC_ALLOC=boot_flatcc_alloc
    FLATCC_FREE=boot_flatcc_free
    FLATCC_REALLOC=boot_flatcc_realloc
    FLATCC_CALLOC=boot_flatcc_calloc
    FLATCC_ALIGNED_ALLOC=boot_flatcc_aligned_alloc
    FLATCC_ALIGNED_FREE=boot_flatcc_aligned_free
    FLATCC_EMITTER_ALLOC=boot_flatcc_alloc
    FLATCC_EMITTER_FREE=boot_flatcc_free)

# LAY-001: BOOT lives in bank 1 sectors 0-1 (32 KB); the linker script
# fails the build if the image grows past 0x08007FFF.
target_link_options(boot.elf PRIVATE
    "-T${CMAKE_SOURCE_DIR}/ld/boot.ld")

# --- Static analysis (BNFR-005) ----------------------------------------------
# clang-tidy reuses the compile flags this very configure step produced, so
# it must run from a configured build tree:  make lint / make lint-strict.
find_program(CLANG_TIDY clang-tidy REQUIRED)
option(BOOT_LINT_STRICT "Treat any clang-tidy finding as an error (lint-strict)" OFF)
if(BOOT_LINT_STRICT)
    set(BOOT_LINT_FLAGS --warnings-as-errors=*)
endif()

# The target runs with the build dir as CWD, so make source paths absolute.
set(BOOT_LINT_SOURCES)
foreach(_src ${PROJECT_SOURCES})
    if(_src MATCHES "\\.c$")
        list(APPEND BOOT_LINT_SOURCES "${CMAKE_CURRENT_SOURCE_DIR}/${_src}")
    endif()
endforeach()
# clang-tidy runs the host clang but needs the newlib headers used by the
# arm-none-eabi toolchain in the container; otherwise <string.h> is missing.
set(BOOT_LINT_EXTRA_ARGS --extra-arg=-isystem
    --extra-arg=/usr/lib/arm-none-eabi/include)

add_custom_target(clang-tidy
    COMMAND ${CLANG_TIDY} -quiet -p ${CMAKE_BINARY_DIR} ${BOOT_LINT_FLAGS}
            ${BOOT_LINT_EXTRA_ARGS} ${BOOT_LINT_SOURCES}
    COMMENT "Running clang-tidy over the project sources"
    WORKING_DIRECTORY ${CMAKE_SOURCE_DIR}
    VERBATIM)

set(ELF "$<TARGET_FILE:boot.elf>")
add_custom_command(TARGET boot.elf POST_BUILD
    COMMAND ${CMAKE_OBJCOPY} -O binary  ${ELF} ${CMAKE_BINARY_DIR}/boot.bin
    COMMAND ${CMAKE_OBJCOPY} -O ihex    ${ELF} ${CMAKE_BINARY_DIR}/boot.hex
    COMMAND ${SIZE_TOOL} $<TARGET_FILE:boot.elf>
    BYPRODUCTS ${CMAKE_BINARY_DIR}/boot.bin ${CMAKE_BINARY_DIR}/boot.hex
    COMMENT "Generating boot.bin and boot.hex")
